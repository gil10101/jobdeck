% ATS-friendly settings

% Uniform section header with a rule
\newcommand{\resumesection}[1]{
    \vspace{6pt}
    {\large\textbf{#1}}\\
    \vspace{3pt}
    \hrule
    \vspace{6pt}
}

% Uniform job/education entry (ends its own paragraph so consecutive entries
% never collide, regardless of what follows)
\newcommand{\resumeentry}[4]{
    \par\textbf{#1} \hfill #2 \\
    \vspace{1pt}
    #3 \hfill #4\par
}

% Uniform project entry
\newcommand{\projectentry}[2]{
    \textbf{#1} \newline #2
    \vspace{2pt}
}

% ATS-friendly uniform bullet points
\newenvironment{resumelist}{
    \begin{itemize}[leftmargin=15pt, topsep=0pt, itemsep=2pt, parsep=0pt]
}{
    \end{itemize}
    \vspace{4pt}
}
